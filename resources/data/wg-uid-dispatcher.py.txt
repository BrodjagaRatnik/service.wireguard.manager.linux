#!/usr/bin/env python3
""" ./resources/data/wg-uid-dispatcher.py.txt """

import json
import os
import subprocess
import sys
import time

CONFIG_PATH = "/etc/wireguard-manager/dispatcher.json"
SETTLE_DELAY_SECONDS = 3.0
RELEVANT_ACTIONS = ("up", "vpn-up")


def load_config():
    defaults = {"enabled": True, "dns_priority": "100", "routing_rule": ""}
    try:
        with open(CONFIG_PATH, "r", encoding="utf-8") as config_file:
            stored = json.load(config_file)
        defaults.update({key: value for key, value in stored.items() if key in defaults})
    except (OSError, ValueError):
        return defaults
    return defaults


def run_nmcli(arguments):
    subprocess.run(
        ["nmcli"] + arguments,
        stdout=subprocess.DEVNULL,
        stderr=subprocess.PIPE,
        check=False
    )


def main():
    interface = sys.argv[1] if len(sys.argv) > 1 else ""
    action = sys.argv[2] if len(sys.argv) > 2 else ""
    connection_id = os.environ.get("CONNECTION_ID", "")

    if action not in RELEVANT_ACTIONS:
        return 0

    config = load_config()
    if not config.get("enabled"):
        return 0

    time.sleep(SETTLE_DELAY_SECONDS)

    if config.get("dns_priority"):
        run_nmcli(["connection", "modify", connection_id,
                   "ipv4.dns-priority", str(config.get("dns_priority"))])

    if config.get("routing_rule"):
        run_nmcli(["connection", "modify", connection_id,
                   "ipv4.routing-rules", str(config.get("routing_rule"))])

    if interface:
        run_nmcli(["device", "reapply", interface])
    return 0


if __name__ == "__main__":
    sys.exit(main())
