#!/usr/bin/env python3
""" ./resources/data/uid_dispatcher_install.py.txt """

import json
import os
import shutil
import stat
import sys

DISPATCH_NAME = "50-wg-uid-restore"
DISPATCH_DIR = "/etc/NetworkManager/dispatcher.d"
CONFIG_DIR = "/etc/wireguard-manager"
CONFIG_NAME = "dispatcher.json"


def build_config(uid):
    return {
        "enabled": True,
        "dns_priority": "100",
        "routing_rule": "priority 20000 not uidrange {0}-{0} table 254".format(uid)
    }


def main():
    uid = os.environ.get("PKEXEC_UID", "")
    if not uid.isdigit():
        sys.stderr.write("Installer: Elevated launch context missing. Use pkexec.\n")
        return 1

    template = sys.argv[1] if len(sys.argv) > 1 else ""
    if not template or not os.path.isfile(template):
        sys.stderr.write("Installer: Dispatcher template argument invalid.\n")
        return 1

    os.makedirs(CONFIG_DIR, exist_ok=True)
    config_path = os.path.join(CONFIG_DIR, CONFIG_NAME)
    if not os.path.exists(config_path):
        with open(config_path, "w", encoding="utf-8") as config_file:
            json.dump(build_config(int(uid)), config_file, indent=4)

    dest_path = os.path.join(DISPATCH_DIR, DISPATCH_NAME)
    shutil.copy2(template, dest_path)
    os.chmod(dest_path, stat.S_IRWXU | stat.S_IRGRP | stat.S_IXGRP)
    os.chown(dest_path, 0, 0)

    try:
        os.remove(os.path.abspath(__file__))
    except OSError:
        pass
    return 0


if __name__ == "__main__":
    sys.exit(main())
